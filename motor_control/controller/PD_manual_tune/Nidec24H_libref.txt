/**
 * @class Nidec24H
 * @brief Main class for controlling the Nidec 24H motor.
 *
 * Encapsulates all functionality for motor control including speed, direction,
 * braking, and encoder reading.
 */
class Nidec24H
{
public:

    /**
     * @brief Constructor for the Nidec24H motor controller.
     *
     * @param brakePin ESP32 pin number for the motor brake control.
     * @param pwmPin ESP32 pin number for the motor PWM speed control.
     * @param dirPin ESP32 pin number for the motor direction control.
     * @param encAPin ESP32 pin number for the encoder's Channel A.
     * @param encBPin ESP32 pin number for the encoder's Channel B.
     * @param pwmFreq (Optional) The frequency for the PWM signal in Hz. Default
     * is 20000.
     * @param pwmResolution (Optional) The resolution for the PWM duty cycle in
     * bits. Default is 8 (0-255).
     * @param pulsesPerRotation (Optional) The number of pulses per revolution
     * (PPR) for the encoder (as per datasheet, not the 4x quadrature count).
     * Default is 100.
     */
    Nidec24H(uint8_t brakePin, uint8_t pwmPin, uint8_t dirPin, uint8_t encAPin, uint8_t encBPin,
             uint32_t pwmFreq = 20000, uint8_t pwmResolution = 8, uint32_t pulsesPerRotation = 100);

    /**
     * @brief Initializes all hardware pins, PWM, and encoder.
     *
     * Sets pin modes, engages the brake, stops the motor (PWM 0),
     * and resets the encoder count to zero.
     */
    void begin();

    /**
     * @brief Engages or releases the motor brake.
     *
     * @param engageBrake true to engage the brake (which also stops PWM),
     * false to release the brake.
     */
    void brake(bool engageBrake);

    /**
     * @brief Sets the motor speed and direction.
     *
     * @param speed A value from -1.0 (max speed clockwise) to +1.0 (max speed
     * counter-clockwise). 0.0 will stop the motor (stops PWM) but will *not*
     * engage the brake.
     */
    void setSpeed(float speed);

    /**
     * @brief Reads the cumulative rotational angle from the encoder.
     *
     * The angle is reported in radians. The angle increases for
     * counter-clockwise (CCW) rotation, consistent with the right-hand rule.
     *
     * @return The cumulative angle in radians.
     */
    double readAngle();

    /**
     * @brief Resets the encoder's internal counter to zero.
     *
     * This effectively sets the current motor position as the new origin (0
     * radians).
     */
    void resetEncoder();

private:
    // Pin assignments
    uint8_t _brakePin; ///< Pin for brake control
    uint8_t _pwmPin;   ///< Pin for PWM speed control
    uint8_t _dirPin;   ///< Pin for direction control
    uint8_t _encAPin;  ///< Pin for encoder channel A
    uint8_t _encBPin;  ///< Pin for encoder channel B

    // PWM configuration
    uint32_t _pwmFreq;           ///< PWM frequency in Hz
    uint8_t _pwmResolution;      ///< PWM resolution in bits
    uint32_t _pulsesPerRotation; ///< Encoder PPR (from constructor)
    uint32_t _countsPerRev;      ///< Full quadrature counts per revolution (PPR * 4)
    uint32_t _maxDuty;           ///< Maximum duty cycle value (2^resolution - 1)

    /// @brief Encoder object instance from ESP32Encoder library.
    ESP32Encoder _encoder;

    /// @brief Static counter to assign unique LEDC channels for older
    /// Arduino-ESP32 cores.
    static uint8_t _channelPoolNext;

    /// @brief The ESP32 LEDC channel used for PWM. -1 if using newer API.
    int8_t _ledcChannel;

    /**
     * @brief Internal helper to attach the PWM pin to the LEDC peripheral.
     *
     * Handles compatibility between Arduino-ESP32 core 2.x and 3.x APIs.
     */
    void _pwmAttach();

    /**
     * @brief Internal helper to write a duty cycle value to the LEDC
     * peripheral.
     *
     * Handles compatibility and ensures duty is capped at _maxDuty.
     * @param duty The raw duty cycle value to write (0 to _maxDuty).
     */
    void _pwmWrite(uint32_t duty);
};

#endif // NIDEC24H_H
